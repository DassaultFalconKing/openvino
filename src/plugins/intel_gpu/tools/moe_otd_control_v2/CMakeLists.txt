cmake_minimum_required(VERSION 3.20)
project(moe_otd_native_controls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(OV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(MOE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../src/graph/impls/ocl_v2/moe")
set(CASES "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/unit/test_cases")
set(gtest_force_shared_crt ON CACHE BOOL "Match MSVC runtime" FORCE)
add_subdirectory("${OV_ROOT}/thirdparty/gtest/gtest" "gtest" EXCLUDE_FROM_ALL)
enable_testing()
add_executable(otd_control_test test_control.cpp)
if(MSVC)
    target_compile_options(otd_control_test PRIVATE /UNDEBUG)
else()
    target_compile_options(otd_control_test PRIVATE -UNDEBUG)
endif()
add_executable(otd_lru_test test_lru.cpp "${MOE_DIR}/lru_cache.cpp")
add_executable(otd_pack_tests "${CASES}/moe_offload_pack_test.cpp"
    "${CASES}/moe_offload_pack_publish_test.cpp" "${MOE_DIR}/moe_otd_pack.cpp")
target_include_directories(otd_pack_tests PRIVATE "${MOE_DIR}/../..")
target_link_libraries(otd_pack_tests PRIVATE gtest_main)
if(MSVC)
    foreach(target otd_control_test otd_lru_test otd_pack_tests)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
    endforeach()
endif()
add_test(NAME controls COMMAND otd_control_test)
add_test(NAME lru COMMAND otd_lru_test)
add_test(NAME pack COMMAND otd_pack_tests)
